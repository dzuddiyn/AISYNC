const ASC_PENDING_KEY = 'asc.pending.fragment.v0.1'; function getAscFragment(hashValue) { const hash = String(hashValue || '').replace(/^#/, ''); if (!hash.startsWith('asc=')) return null; const payload = hash.slice(4); return payload ? '#asc=' + payload : null; } function preservePendingRequest(locationLike, storage) { const current = getAscFragment(locationLike.hash); if (current) { storage.setItem(ASC_PENDING_KEY, current); return { fragment: current, source: 'url' }; } const saved = storage.getItem(ASC_PENDING_KEY); const restored = getAscFragment(saved); if (restored) { return { fragment: restored, source: 'session' }; } return { fragment: null, source: 'none' }; } function base64UrlToUtf8(value) { if (!/^[A-Za-z0-9_-]+$/.test(value)) { throw new Error('ASC fragment payload is not valid Base64URL.'); } const padded = value.replace(/-/g, '+').replace(/_/g, '/') .padEnd(Math.ceil(value.length / 4) * 4, '='); const binary = atob(padded); const bytes = Uint8Array.from(binary, ch => ch.charCodeAt(0)); return new TextDecoder().decode(bytes); } function decodePendingEnvelope(fragment) { const normalized = getAscFragment(fragment); if (!normalized) throw new Error('ASC pending fragment is missing.'); const encoded = normalized.slice('#asc='.length); const envelope = JSON.parse(base64UrlToUtf8(encoded)); if (envelope?.envelope_version !== '0.1') { throw new Error('Unsupported ASC envelope version.'); } return envelope; } function renderPendingPreview(envelope) { const preview = document.getElementById('preview'); const payload = document.getElementById('payload'); payload.textContent = JSON.stringify(envelope.contract, null, 2); preview.hidden = false; } function setStatus(message, isError) { const status = document.getElementById('status'); status.textContent = message; status.className = 'card' + (isError ? ' error' : ''); } function getAppsScriptLocation(callback) { if ( typeof google !== 'undefined' && google.script && google.script.url && typeof google.script.url.getLocation === 'function' ) { google.script.url.getLocation(function(location) { callback({ hash: location.hash || '' }); }); return; } // Fallback for local/non-Apps-Script tests. callback(window.location); } const ASC_CONFIRM_ACTION = 'CONFIRM_AND_SYNC'; const ASC_MAIN_UI_PREFIX = 'https://sites.google.com/'; let ascPendingState = null; // Confirmation is created only from an explicit user click and is bound to one request. function buildConfirmation(requestId) { return { confirmed: true, action: ASC_CONFIRM_ACTION, requestId: requestId }; } // Redirect only after a verified SUCCESS receipt and persisted HISTORY. Everything else stays here. function canRedirectAfterSync(result) { return Boolean(result) && result.state === 'SYNCED' && typeof result.redirect === 'string' && result.redirect.indexOf(ASC_MAIN_UI_PREFIX) === 0 && result.redirect.length > ASC_MAIN_UI_PREFIX.length && !/\s/.test(result.redirect) && Boolean(result.receipt) && result.receipt.status === 'SUCCESS' && result.receipt.verified === true && result.historyOutcome === 'HISTORY_PERSISTED'; } function describeSyncResult(result) { if (!result || typeof result !== 'object') { return { isError: true, message: 'FAILED — no factual sync result was returned. Nothing is reported as saved.' }; } if (canRedirectAfterSync(result)) { return { isError: false, message: 'SUCCESS — verified write and HISTORY recorded. Returning to the main ASC UI…' }; } if (result.state === 'SYNCED_REDIRECT_UNAVAILABLE') { return { isError: false, message: 'SUCCESS — verified write and HISTORY recorded. Main ASC UI link is not configured.' }; } if (result.state === 'AWAITING_CONFIRMATION') { return { isError: true, message: 'Not saved — explicit CONFIRM & SYNC is required.' }; } const code = result.error && result.error.code ? result.error.code : 'UNKNOWN'; const reason = result.error && result.error.message ? result.error.message : 'Sync did not complete.'; return { isError: true, message: 'FAILED (' + code + ') — ' + reason }; } function renderSyncResult(result, deps) { const doc = (deps && deps.document) || document; const view = describeSyncResult(result); const status = doc.getElementById('status'); status.textContent = view.message; status.className = 'card' + (view.isError ? ' error' : ''); const receiptCard = doc.getElementById('receipt'); const receiptPre = doc.getElementById('receipt-json'); if (result && result.receipt) { receiptPre.textContent = JSON.stringify(result.receipt, null, 2); receiptCard.hidden = false; } const button = doc.getElementById('confirm-sync'); if (canRedirectAfterSync(result)) { if (button) button.disabled = true; const storage = (deps && deps.storage) || window.sessionStorage; storage.removeItem(ASC_PENDING_KEY); const navigate = (deps && deps.navigate) || function(url) { window.top.location.href = url; }; const link = doc.getElementById('main-ui-link'); if (link) { link.href = result.redirect; link.hidden = false; } navigate(result.redirect); return 'REDIRECTED'; } // Failure or incomplete: keep the pending request so the owner can see and retry it. if (button) button.disabled = !(ascPendingState && ascPendingState.writeEnabled); return 'STAYED'; } // The server may settle asynchronously; a RESULT_PENDING reply is followed by one factual // result fetch. An unavailable result is shown as FAILED/unknown, never as success. function handleConfirmResponse(result, deps) { const run = (deps && deps.run) || (typeof google !== 'undefined' && google.script && google.script.run); const render = (deps && deps.render) || renderSyncResult; if (result && result.state === 'RESULT_PENDING' && typeof result.requestId === 'string' && run) { run .withSuccessHandler(function(final) { render(final); }) .withFailureHandler(function(error) { render({ state: 'FAILED', error: { code: 'RESULT_CALL_FAILED', message: 'Sync result could not be fetched. Check HISTORY; nothing is reported as saved.' } }); }) .getConfirmSyncResult(result.requestId); return 'FETCHING_RESULT'; } render(result); return 'RENDERED'; } function requestConfirmAndSync() { if (!ascPendingState || !ascPendingState.writeEnabled) { renderSyncResult({ state: 'FAILED', error: { code: 'SYNC_DISABLED', message: 'Sync is not enabled on this deployment.' } }); return; } const button = document.getElementById('confirm-sync'); button.disabled = true; setStatus('Syncing… write is being attempted and verified. Do not treat this as saved yet.', false); google.script.run .withSuccessHandler(handleConfirmResponse) .withFailureHandler(function(error) { renderSyncResult({ state: 'FAILED', error: { code: 'SERVER_CALL_FAILED', message: String(error && error.message || error) } }); }) .confirmAndSync({ fragment: ascPendingState.fragment, confirmation: buildConfirmation(ascPendingState.requestId) }); } function showConfirmControls(envelope, fragment, bootstrap) { ascPendingState = { fragment: fragment, requestId: envelope.request_id, writeEnabled: Boolean(bootstrap && bootstrap.writeEnabled === true) }; const controls = document.getElementById('confirm-controls'); const button = document.getElementById('confirm-sync'); const note = document.getElementById('confirm-note'); controls.hidden = false; button.disabled = !ascPendingState.writeEnabled; note.textContent = ascPendingState.writeEnabled ? Preview only — nothing will be saved.'; button.onclick = requestConfirmAndSync; } function loadBootstrap(callback) { if (typeof google !== 'undefined' && google.script && google.script.run) { google.script.run .withSuccessHandler(callback) .withFailureHandler(function() { callback({ writeEnabled: false }); }) .getBootstrapState(); return; } callback({ writeEnabled: false }); } function bootAscPreview() { getAppsScriptLocation(function(locationLike) { const pending = preservePendingRequest(locationLike, window.sessionStorage); if (!pending.fragment) { setStatus('Signed in. Pending request restored from ' + pending.source + '. Preview only — no write has occurred.', false ); loadBootstrap(function(bootstrap) { showConfirmControls(envelope, pending.fragment, bootstrap); }); } catch (error) { setStatus('Pending request could not be decoded: ' + error.message, true); } }); } if (typeof window !== 'undefined' && typeof window.addEventListener === 'function') { window.addEventListener('DOMContentLoaded', bootAscPreview); }
