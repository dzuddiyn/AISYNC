const ASC_PENDING_KEY = 'asc.pending.fragment.v0.1'; function getAscFragment(hashValue) { const hash = String(hashValue || '').replace(/^#/, ''); if (!hash.startsWith('asc=')) return null; const payload = hash.slice(4); return payload ? '#asc=' + payload : null; } function preservePendingRequest(locationLike, storage) { const current = getAscFragment(locationLike.hash); if (current) { storage.setItem(ASC_PENDING_KEY, current); return { fragment: current, source: 'url' }; } const saved = storage.getItem(ASC_PENDING_KEY); const restored = getAscFragment(saved); if (restored) { return { fragment: restored, source: 'session' }; } return { fragment: null, source: 'none' }; } function base64UrlToUtf8(value) { if (!/^[A-Za-z0-9_-]+$/.test(value)) { throw new Error('ASC fragment payload is not valid Base64URL.'); } const padded = value.replace(/-/g, '+').replace(/_/g, '/') .padEnd(Math.ceil(value.length / 4) * 4, '='); const binary = atob(padded); const bytes = Uint8Array.from(binary, ch => ch.charCodeAt(0)); return new TextDecoder().decode(bytes); } function decodePendingEnvelope(fragment) { const normalized = getAscFragment(fragment); if (!normalized) throw new Error('ASC pending fragment is missing.'); const encoded = normalized.slice('#asc='.length); const envelope = JSON.parse(base64UrlToUtf8(encoded)); if (envelope?.envelope_version !== '0.1') { throw new Error('Unsupported ASC envelope version.'); } return envelope; } function renderPendingPreview(envelope) { const preview = document.getElementById('preview'); const payload = document.getElementById('payload'); payload.textContent = JSON.stringify(envelope.contract, null, 2); preview.hidden = false; } function setStatus(message, isError) { const status = document.getElementById('status'); status.textContent = message; status.className = 'card' + (isError ? ' error' : ''); } function showNoPendingRequest(deps) { const doc = (deps && deps.document) || document; setSaveState( 'NO REQUEST', 'Signed in. No pending ASC request found. Return to the original ASC Public Front Door tab and press CONTINUE to replay the preserved request.', false, { document: doc } ); } function setSaveState(state, message, isError, deps) { const doc = (deps && deps.document) || document; const stateEl = doc.getElementById('save-state'); if (stateEl) { stateEl.textContent = state; stateEl.className = 'badge save-state' + (isError ? ' error' : ''); } const status = doc.getElementById('status'); if (status && typeof message === 'string') { status.textContent = message; status.className = 'card' + (isError ? ' error' : ''); } } function getAppsScriptLocation(callback) { if ( typeof google !== 'undefined' && google.script && google.script.url && typeof google.script.url.getLocation === 'function' ) { google.script.url.getLocation(function(location) { callback({ hash: location.hash || '' }); }); return; } // Fallback for local/non-Apps-Script tests. callback(window.location); } const ASC_CONFIRM_ACTION = 'CONFIRM_AND_SYNC'; const ASC_MAIN_UI_PREFIX = 'https://sites.google.com/'; // Pre-claim failures where the same request_id was never consumed. const ASC_RETRYABLE_CODES = ['RESULT_CACHE_UNAVAILABLE']; let ascPendingState = null; // Confirmation is created only from an explicit user click and is bound to one request. function buildConfirmation(requestId) { return { confirmed: true, action: ASC_CONFIRM_ACTION, requestId: requestId }; } // Redirect only after a verified SUCCESS receipt and persisted HISTORY. Everything else stays here. function isFactualSavedResult(result) { return Boolean(result) && (result.state === 'SYNCED' || result.state === 'SYNCED_REDIRECT_UNAVAILABLE') && Boolean(result.receipt) && result.receipt.status === 'SUCCESS' && result.receipt.verified === true && result.historyOutcome === 'HISTORY_PERSISTED'; } // Redirect only after a factual SAVED state. Everything else stays here. function canRedirectAfterSync(result) { return isFactualSavedResult(result) && result.state === 'SYNCED' && typeof result.redirect === 'string' && result.redirect.indexOf(ASC_MAIN_UI_PREFIX) === 0 && result.redirect.length > ASC_MAIN_UI_PREFIX.length && !/\s/.test(result.redirect); } function classifySaveState(result) { if (isFactualSavedResult(result)) return 'SAVED'; if (result && result.state === 'RESULT_PENDING') return 'SYNCING'; if (result && result.state === 'AWAITING_CONFIRMATION') return 'UNSAVED'; return 'FAILED'; } function describeSyncResult(result) { const state = classifySaveState(result); if (state === 'SAVED') { if (canRedirectAfterSync(result)) { return { state, isError: false, message: 'SAVED — verified persistence and HISTORY recorded. Returning to the main ASC UI…' }; } return { state, isError: false, message: 'SAVED — verified persistence and HISTORY recorded. Main ASC UI link is not configured.' }; } if (state === 'SYNCING') { return { state, isError: false, message: 'SYNCING — persistence is still being resolved. Do not treat this as saved yet.' }; } if (state === 'UNSAVED') { return { state, isError: false, message: 'UNSAVED — explicit CONFIRM & SYNC is required before persistence.' }; } if (!result || typeof result !== 'object') { return { state: 'FAILED', isError: true, message: 'FAILED — no factual sync result was returned. Nothing is reported as saved.' }; } const code = result.error && result.error.code ? result.error.code : 'UNKNOWN'; const reason = result.error && result.error.message ? result.error.message : 'Sync did not complete.'; return { state: 'FAILED', isError: true, message: 'FAILED (' + code + ') — ' + reason }; } function describeReceipt(result) { if (!result || !result.receipt) return ''; const receipt = result.receipt; const lines = []; lines.push('State: ' + classifySaveState(result)); lines.push('Outcome: ' + (receipt.adapter_outcome || 'Not provided')); lines.push('Resource: ' + (receipt.affected_resource || 'Not provided')); if (receipt.commit_or_record_id) { lines.push('Commit / record: ' + receipt.commit_or_record_id); } else if (receipt.adapter_outcome === 'NO_CHANGE') { lines.push('Commit / record: No new commit — desired content already matched the authoritative target.'); } else if (receipt.adapter_outcome === 'VERIFIED_WRITE_RECONCILED') { lines.push('Commit / record: No attributable commit SHA — persisted content was verified by reconciliation.'); } else { lines.push('Commit / record: Not provided'); } const performed = receipt.write_performed === true ? An unavailable result is shown as FAILED/unknown, never as success. function handleConfirmResponse(result, deps) { const run = (deps && deps.run) || (typeof google !== 'undefined' && google.script && google.script.run); const render = (deps && deps.render) || renderSyncResult; if (result && result.state === 'RESULT_PENDING' && typeof result.requestId === 'string' && run) { run .withSuccessHandler(function(final) { render(final); }) .withFailureHandler(function(error) { render({ state: 'FAILED', error: { code: 'RESULT_CALL_FAILED', message: 'Sync result could not be fetched. Check HISTORY; nothing is reported as saved.' } }); }) .getConfirmSyncResult(result.requestId); return 'FETCHING_RESULT'; } render(result); return 'RENDERED'; } function requestConfirmAndSync() { if (!ascPendingState || !ascPendingState.writeEnabled) { renderSyncResult({ state: 'FAILED', error: { code: 'SYNC_DISABLED', message: 'Sync is not enabled on this deployment.' } }); return; } const button = document.getElementById('confirm-sync'); button.disabled = true; setSaveState('SYNCING', 'SYNCING — write is being attempted and verified. CONFIRM & SYNC is enabled only after the // server preview check (structure, expiry, SHA-256 integrity, owner) for this exact request. function describeServerPreview(serverPreview, envelope) { const valid = Boolean(serverPreview) && serverPreview.securityValid === true && Boolean(envelope) && serverPreview.requestId === envelope.request_id; if (!valid) { const code = serverPreview && serverPreview.error && serverPreview.error.code ? serverPreview.error.code : 'SERVER_PREVIEW_UNAVAILABLE'; const reason = serverPreview && serverPreview.error && serverPreview.error.message ? serverPreview.error.message : 'Server security check did not pass.'; return { enabled: false, isError: true, note: 'REJECTED (' + code + ') — ' + reason + ' CONFIRM & SYNC is disabled; nothing will be saved.' }; } if (serverPreview.writeEnabled !== true) { return { enabled: false, isError: false, note: 'Request is security-valid, but CONFIRM & SYNC is not enabled on this deployment.
